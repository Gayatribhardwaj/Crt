1. WAP that takes input from user that print all number one to n that is divisible by two.

#include<iostream>
using namespace std;

int main() {
    int n ;
    cout<<"Enter n value : ";
    cin>>n;
    
    for (int i = 1; i<=n; i++ ) {

        if(i%2==0) {
            cout<<"Print the number"<<i<<endl;
        }
    }
}



2. WAP that takes continuously integer as a input from user until user giver -ve number. print the all positive number sum.

#include <iostream>
using namespace std;

int main() {
    int num;
    int sum = 0;

    while (true) {
        cout << "Enter a number: ";
        cin >> num;

        if (num < 0) {
            break;  // stop when negative number is entered
        }

        sum += num;  // add positive numbers
    }

    cout << "Sum of positive numbers = " << sum;

    return 0;
}


3 WAP that takes an positive integer n from the user you have to calculate the factorial of n .

#include<iostream>
using namespace std;

int main() {
    int n;
    cout<<"Enter the value of n : ";
    cin>>n;
    
    int f=1;
    int i=1;
    
    while(i<=n){
        f=f*i;
        i=i+1;
    }
    cout<<"Factorial : "<<f;
}
